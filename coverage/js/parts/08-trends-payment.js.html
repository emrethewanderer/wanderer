export async function doForgotPassword() {
  const email = document.getElementById('login-email').value.trim();
  if (!email) { document.getElementById('auth-error').textContent = t('auth.enter_email_first'); return; }
  const btn = document.getElementById('login-btn');
  btn.disabled = true; btn.textContent = '...';
  const { error } = await sb.auth.resetPasswordForEmail(email, {
    redirectTo: window.location.href
  });
  btn.disabled = false; btn.textContent = t('auth.btn.login_short');
  if (error) { document.getElementById('auth-error').textContent = trAuthErr(error.message); }
  else {
    const errEl = document.getElementById('auth-error');
    errEl.style.color = 'var(--gold)';
    errEl.textContent = t('auth.reset_sent');
    // 5 saniye sonra stili otomatik sıfırla
    setTimeout(() => { errEl.style.color = ''; errEl.textContent = ''; }, 5000);
  }
}

export function filterHistory(q) {
  const list = document.getElementById('full-history-list');
  list.innerHTML = '';
  const lq = q.toLowerCase().trim();
  const sortedSids = Object.keys(S.allSessions).sort((a,b) =>
    getSessionLastActivity(b) - getSessionLastActivity(a));
  const filtered = sortedSids.filter(sid => {
    if (!lq) return true;
    return S.allSessions[sid].some(m => m.content && m.content.toLowerCase().includes(lq));
  });
  if (!filtered.length) { list.innerHTML = '<div class="empty-state" style="padding:32px 28px;">' + t('ui.no_results') + '</div>'; return; }
  filtered.forEach(sid => {
    const msgs  = S.allSessions[sid];
    const first = msgs.find(m => m.role === 'user');
    if (!first) return;
    const short = first.content.length > 55 ? first.content.slice(0, 55) + '…' : first.content;
    const safeShort = escapeHTML(short);
    const dStr  = new Date(first.created_at).toLocaleDateString(getCurrentLanguage() || 'tr', { day: 'numeric', month: 'long', year: 'numeric' });
    const item  = document.createElement('div');
    item.className = 'history-item';
    item.onclick = () => {
      S.currentSessId = sid;
      S.chatHistory   = msgs.map(m => ({ role: m.role, content: m.content, mode: m.mode || '' }));
      S.summaryInProgress = false;
      document.getElementById('messages-area').innerHTML = '';
      S.chatHistory.forEach(m => appendMsg(m.role === 'user' ? 'user' : 'coach', m.content, m.mode || ''));
      updateSessionRing();
      EventBus.emit('navigate', { view: 'chat' });
      applySessionPartDots(sid);
    };
    item.innerHTML = `<div class="h-date">${dStr}</div><div class="h-title">${safeShort}</div>`;
    list.appendChild(item);
  });
}

export function renderAvoidanceCloud() {
  const el = document.getElementById('avoidance-cloud');
  if (!el) return;
 
  // Tüm kullanıcı mesajlarını tara
  const allUserMsgs = getAllMessages()
    .filter(m => m.role === 'user')
    .map(m => m.content || '');
 
  const counts = {};
  const labels = {
    'bilmiyorum':    'Bilmiyorum',
    'belki':         'Belki',
    'sanırım':       'Sanırım',
    'zor':           'Zor',
    'yapamam':       'Yapamam',
    'sonra':         'Sonra',
    'şimdi değil':   'Şimdi Değil',
    'fark etmez':    'Fark Etmez',
    'önemli değil':  'Önemli Değil',
    'neyse':         'Neyse',
    'yorgunum':      'Yorgunum',
    'motivasyon':    'Motivasyon Yok',
    'zamanım yok':   'Zamanım Yok',
    'hazır değil':   'Hazır Değil',
    'olmaz':         'Olmaz',
  };
 
  Object.keys(labels).forEach(k => {
    const re = new RegExp(k, 'gi');
    allUserMsgs.forEach(msg => {
      const m = msg.match(re);
      if (m) counts[k] = (counts[k] || 0) + m.length;
    });
  });
 
  const sorted = Object.entries(counts).sort((a,b) => b[1]-a[1]).slice(0, 12);
  if (!sorted.length) {
    el.innerHTML = '<div style="font-size:12px;color:var(--text-dim);font-style:italic;">' + t('ui.no_avoidance', 'Henüz kaçınma kalıbı tespit edilmedi.') + '</div>';
    return;
  }
  const max = sorted[0][1];
  el.innerHTML = sorted.map(([k, c]) => {
    const isHot = c >= max * 0.6;
    return `<span class="avoid-tag ${isHot?'hot':''}">${labels[k]}<span class="tag-count">${c}×</span></span>`;
  }).join('');
}

export function renderAvoidanceTrend() {
  const ctx = document.getElementById('avoidanceTrendChart');
  if (!ctx) return;
 
  // Son 8 haftayı hesapla
  const now = new Date();
  const weeks = [];
  for (let i = 7; i >= 0; i--) {
    const weekStart = new Date(now);
    weekStart.setDate(now.getDate() - (i * 7) - now.getDay());
    weekStart.setHours(0, 0, 0, 0);
    const weekEnd = new Date(weekStart);
    weekEnd.setDate(weekStart.getDate() + 7);
 
    const label = weekStart.toLocaleDateString(getCurrentLanguage() || 'tr', { day: 'numeric', month: 'short' });
 
    // O haftaki kullanıcı mesajlarında kaçınma kalıplarını say
    const weekMsgs = getAllMessages()
      .filter(m => m.role === 'user' && m.created_at)
      .filter(m => {
        const d = new Date(m.created_at);
        return d >= weekStart && d < weekEnd;
      })
      .map(m => m.content || '');
 
    let avoidCount = 0;
    const patterns = [
      /bilmiyorum/gi, /emin değil/gi, /belki/gi, /sanırım/gi, /ne bileyim/gi,
      /yapamam/gi, /mümkün değil/gi, /sonra/gi, /şimdi değil/gi,
      /fark etmez/gi, /önemli değil/gi, /neyse/gi, /olmaz/gi,
      /yorgunum/gi, /zamanım yok/gi, /hazır değil/gi
    ];
    weekMsgs.forEach(msg => {
      patterns.forEach(p => {
        const m = msg.match(p);
        if (m) avoidCount += m.length;
      });
    });
 
    weeks.push({ label, count: avoidCount });
  }
 
  const labels = weeks.map(w => w.label);
  const values = weeks.map(w => w.count);
  const maxVal = Math.max(...values, 1);
 
  // Renk: düşük = yeşil, yüksek = kırmızı
  const colors = values.map(v => {
    const ratio = v / maxVal;
    if (ratio < 0.3) return 'rgba(91,185,123,0.7)';
    if (ratio < 0.6) return 'rgba(184,149,60,0.7)';
    return 'rgba(192,57,43,0.7)';
  });
 
  if (avoidanceTrendChartObj) avoidanceTrendChartObj.destroy();
  avoidanceTrendChartObj = new Chart(ctx, {
    type: 'bar',
    data: {
      labels,
      datasets: [{
        data: values,
        backgroundColor: colors,
        borderWidth: 0,
        borderRadius: 2,
      }]
    },
    options: {
      responsive: true, maintainAspectRatio: false,
      plugins: { legend: { display: false }, tooltip: {
        callbacks: { label: ctx => ctx.parsed.y + ' ' + t('ui.avoidance_word') }
      }},
      scales: {
        y: {
          min: 0,
          grid: { color: 'rgba(255,255,255,0.04)', drawBorder: false },
          ticks: { color: '#52504A', stepSize: 1, precision: 0 }
        },
        x: { grid: { display: false, drawBorder: false }, ticks: { color: '#52504A', font: { size: 10 } } }
      }
    }
  });
}
 
/* ─── Count-up animasyonu yardımcısı ─── */
function maCountUp(el, target, duration = 900, suffix = '') {
  if (!el) return;
  const isFloat = String(target).includes('.');
  const start = 0;
  const startTime = performance.now();
  el.classList.add('counting');
  function step(now) {
    const elapsed = now - startTime;
    const progress = Math.min(elapsed / duration, 1);
    const ease = 1 - Math.pow(1 - progress, 3); // cubic ease-out
    const current = start + (target - start) * ease;
    el.textContent = isFloat ? current.toFixed(1) + suffix : Math.round(current) + suffix;
    if (progress < 1) requestAnimationFrame(step);
    else {
      el.textContent = isFloat ? target.toFixed(1) + suffix : Math.round(target) + suffix;
      el.classList.remove('counting');
      el.classList.add('shine');
      setTimeout(() => el.classList.remove('shine'), 800);
    }
  }
  requestAnimationFrame(step);
}

/* Kullanıcı dostu hata mesajı dönüştürücü — teknik mesajları gizler */
export function userFriendlyError(e) {
  const raw = (e?.message || String(e || '')).toLowerCase();
  if (!raw) return t('error.unexpected');
  if (raw.includes('network') || raw.includes('failed to fetch') || raw.includes('load failed')) {
    return t('error.network');
  }
  if (raw.includes('quota') || raw.includes('exceeded') || raw.includes('rate limit') || raw.includes('429')) {
    return t('error.rate_limit');
  }
  if (raw.includes('401') || raw.includes('unauthorized') || raw.includes('api key')) {
    return t('error.auth');
  }
  if (raw.includes('timeout') || raw.includes('timed out')) {
    return t('error.timeout');
  }
  if (raw.includes('duplicate') || raw.includes('unique constraint')) {
    return t('error.duplicate');
  }
  // Supabase/Postgres spesifik
  if (raw.includes('row-level security') || raw.includes('permission denied')) {
    return t('error.permission');
  }
  // Fallback: genel mesaj
  return t('error.generic');
}
 
export function toastError(e, prefix = '') {
  const msg = prefix ? `${prefix}: ${userFriendlyError(e)}` : userFriendlyError(e);
  showToast(msg, true);
  // Geliştirici için console'a gerçek hatayı bırak
  if (e) console.warn('Hata:', e);
}

export const PRICE_USD = 9;
const PRICING_CACHE_KEY = 'etw_pricing_cache';
const PRICING_CACHE_TTL = 60 * 60 * 1000; // 1 saat
let _pricingRefreshTimer = null;
 
export async function fetchPricingData() {
  // Cache kontrolü
  try {
    const cached = SafeStorage.get(PRICING_CACHE_KEY);
    if (cached && Date.now() - cached.ts < PRICING_CACHE_TTL) return cached;
  } catch (_) {}
 
  // IP'ye göre para birimi
  const geoRes = await fetch('https://ipapi.co/json/');
  if (!geoRes.ok) return null;
  const geo = await geoRes.json();
  const currency = geo.currency;
  if (!currency || currency === 'USD') return null;
 
  // Güncel kur (open.er-api.com — XE uyumlu, ücretsiz)
  const rateRes = await fetch('https://open.er-api.com/v6/latest/USD');
  if (!rateRes.ok) return null;
  const rateData = await rateRes.json();
  const rate = rateData.rates?.[currency];
  if (!rate) return null;
 
  // Ülke locale'ini türet (ör. TR → tr-TR) — sembol doğru çıksın
  const countryCode = geo.country_code || '';
  const langCode = (geo.languages || navigator.language || 'en').split(',')[0].split('-')[0];
  const locale = countryCode ? `${langCode}-${countryCode}` : navigator.language;
 
  const payload = { currency, rate, locale, ts: Date.now() };
  SafeStorage.set(PRICING_CACHE_KEY, payload);
  return payload;
}
 
function applyPricingToDOM(data) {
  const usdEl   = document.getElementById('sub-price-usd');
  const localEl = document.getElementById('sub-price-local');
  if (!usdEl || !localEl) return;
 
  if (!data) return; // USD ülkesi veya hata — $9/ay olduğu gibi kalır
 
  const localAmount = Math.round(PRICE_USD * data.rate);
  const formatted = new Intl.NumberFormat(data.locale || navigator.language, {
    style: 'currency',
    currency: data.currency,
    maximumFractionDigits: 0
  }).format(localAmount);
 
  usdEl.innerHTML = `≈ ${formatted}<sub>/ay</sub>`;
  localEl.textContent = `($${PRICE_USD})`;
}
 
export async function initPricing() {
  try {
    const data = await fetchPricingData();
    applyPricingToDOM(data);
  } catch (_) {}
 
  // Saatlik otomatik güncelleme — sadece bir tane timer çalışsın
  if (!_pricingRefreshTimer) {
    _pricingRefreshTimer = setInterval(async () => {
      // Cache'i geçersiz kıl, taze veri çek
      SafeStorage.remove(PRICING_CACHE_KEY);
      try {
        const data = await fetchPricingData();
        applyPricingToDOM(data);
      } catch (_) {}
    }, PRICING_CACHE_TTL);
  }
}

function wrapText(ctx, text, maxWidth, fontSize) {
  const words = text.split(' ');
  const lines = [];
  let current = '';
  words.forEach(word => {
    const test = current ? current + ' ' + word : word;
    if (ctx.measureText(test).width > maxWidth && current) {
      lines.push(current);
      current = word;
    } else {
      current = test;
    }
  });
  if (current) lines.push(current);
  return lines;
}

/* ── DUYGU TONU → FLUX PROMPT ÜRETICI ── */
function buildImagePrompt(title, summary) {
  // Seans tonunu analiz et
  const text = (title + ' ' + summary).toLowerCase();
 
  // Duygu tonu tespiti
  const isBreakthrough = /yüzleş|gerçek|kabul|fark et|ilk kez|dönüşüm|kırdı|açıldı/.test(text);
  const isResistance   = /kaçış|direniş|kaçtı|direndi|ret|erteledi|olmaz|yapamam/.test(text);
  const isVulnerable   = /ağladı|çaresiz|yalnız|korku|üzgün|kırılgan|bunaldı/.test(text);
  const isAwareness    = /farkında|gördü|anladı|kavradı|kalıp|örüntü|tekrar/.test(text);
 
  // Temel görsel dil: soyut, sinematik, obsidyen
  const base = 'abstract generative art, cinematic chiaroscuro, obsidian black background, ultra high quality, 4k';
 
  if (isBreakthrough) {
    return `${base}, golden light fractal breaking through dark stone, transformation moment, emerald particles emerging from darkness, baroque atmosphere, single ray of warm gold light piercing black void, dust motes suspended in light`;
  }
  if (isResistance) {
    return `${base}, dark geometric labyrinth, deep crimson and charcoal tones, walls of obsidian, maze architecture, heavy shadows, no exit visible, oppressive but beautiful, architectural surrealism`;
  }
  if (isVulnerable) {
    return `${base}, soft silver moonlight on dark water, minimal reflection, single candle flame in vast darkness, fragile warmth, muted gold and deep blue, melancholic beauty, Edward Hopper atmosphere`;
  }
  if (isAwareness) {
    return `${base}, ancient star map on dark stone, constellation patterns in gold and copper, cartographic circles, old manuscript aesthetic, amber bioluminescence, pattern recognition visualized`;
  }
  // Varsayılan: nötr dönüşüm
  return `${base}, abstract smoke and light in dark void, gold and silver tendrils, movement frozen in time, contemplative atmosphere, Zdzisław Beksiński inspired, monumental and quiet`;
}
 
/* ── TOGETHER AI FLUX GÖRSEL ÜRETICI (Edge Function üzerinden) ── */
export async function generateTransformationImage(title, summary) {
  const prompt = buildImagePrompt(title, summary);
 
  const { data: sessionData } = await sb.auth.getSession();
  const accessToken = sessionData?.session?.access_token;
  if (!accessToken) {
    console.warn('Oturum yok — görsel üretim atlanıyor');
    return null;
  }
 
  try {
    const res = await fetch(`${EDGE_FN_BASE}/image-gen`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${accessToken}`,
        'apikey':        SUPABASE_ANON,
        'Content-Type':  'application/json'
      },
      body: JSON.stringify({ prompt })
    });
 
    if (!res.ok) {
      const errText = await res.text().catch(() => '');
      // 500 = server secret eksik → görseli atla, seansı bozma
      if (res.status === 500) {
        console.warn('Görsel servisi yapılandırılmamış — atlanıyor');
        return null;
      }
      throw new Error(`Görsel API hatası (${res.status}): ${errText.slice(0, 200) || res.statusText}`);
    }
 
    const data = await res.json();
    if (data.error)                   throw new Error(data.error.message || JSON.stringify(data.error));
    if (!data.data?.[0]?.b64_json)   throw new Error('Görsel verisi boş döndü');
 
    return 'data:image/png;base64,' + data.data[0].b64_json;
  } catch (e) {
    console.error('Görsel üretim hatası:', e);
    return null;
  }
}
 
/* ── SUPABASE'E GÖRSEL UPLOAD + KART KAYDET ── */
export async function saveTransformationCard(title, summary, imageDataUrl) {
  let imageUrl = null;
 
  // Supabase Storage'a yükle (bucket: transformation-cards, public)
  if (imageDataUrl) {
    try {
      // Base64'ü Blob'a çevir
      const base64 = imageDataUrl.split(',')[1];
      const byteArr = Uint8Array.from(atob(base64), ch => ch.charCodeAt(0));
      const blob    = new Blob([byteArr], { type: 'image/png' });
      const path    = `cards/${S.currentUser.id}/${S.currentSessId}.png`;
 
      const { data: upData, error: upErr } = await sb.storage
        .from('transformation-cards')
        .upload(path, blob, { upsert: true, contentType: 'image/png' });
 
      if (!upErr && upData) {
        const { data: urlData } = sb.storage
          .from('transformation-cards')
          .getPublicUrl(path);
        imageUrl = urlData?.publicUrl || null;
      }
    } catch (e) {
      console.warn('Storage upload hatası:', e.message);
      // Yedek: base64'ü doğrudan kaydet (küçük görseller için)
      imageUrl = imageDataUrl;
    }
  }
 
  // DB'ye kaydet
  try {
    const { error } = await sb.from('transformation_cards').insert([{
      user_id:    S.currentUser.id,
      session_id: S.currentSessId,
      title,
      summary,
      image_url:  imageUrl,
      tone:       buildImagePrompt(title, summary).split(',')[0] // Ton etiketi
    }]);
    if (error) console.warn('Kart kayıt hatası:', error.message);
  } catch (e) {
    console.error('Kart DB hatası:', e);
  }
}

      .order('created_at', { ascending: false });

      const dStr = new Date(card.created_at).toLocaleDateString(getCurrentLanguage() || 'tr', { day: 'numeric', month: 'long' });

          <img src="${card.image_url.replace(/"/g,'&quot;')}" alt="${safeTitle}" loading="lazy">

            <div class="tx-card-title">${safeTitle}</div>

          <div class="tx-card-overlay" style="background:linear-gradient(to top,rgba(0,0,0,0.9) 0%,rgba(0,0,0,0.4) 100%);top:0;">
            <div class="tx-card-date">${dStr}</div>
            <div class="tx-card-title">${safeTitle}</div>
          </div>
        </div>`;
      }
    }).join('');
  } catch (e) {
    console.error('loadTransformationCards hatası:', e);
    grid.innerHTML = '<div class="empty-state" style="grid-column:1/-1;color:var(--red);">' + t('ui.loading_failed') + '</div>';

  if (navigator.share) {

    catch (e) { if (e.name === 'AbortError') return; }
  }
  try {
    await navigator.clipboard.writeText(shareText);
    showToast(t('toast.card_copied'));
  } catch { showToast(t('toast.share_failed'), true); }
}
